Attribute VB_Exposed = False
Attribute VB_TemplateDerived = False
Attribute VB_Customizable = False
Private Sub UserForm_Initialize()
    UpdateCalendarDisplay
End Sub

Private Sub cmdPrev_Click()
    SelMonth = SelMonth - 1
    If SelMonth < 1 Then
        SelMonth = 12
        SelYear = SelYear - 1
    End If
    UpdateCalendarDisplay
End Sub

Private Sub cmdNext_Click()
    SelMonth = SelMonth + 1
    If SelMonth > 12 Then
        SelMonth = 1
        SelYear = SelYear + 1
    End If
    UpdateCalendarDisplay
End Sub

Private Sub cmdToday_Click()
    SelYear = Year(Date)
    SelMonth = Month(Date)
    SelectDate Day(Date)
End Sub

Sub UpdateCalendarDisplay()
    lblTitle.caption = SelYear & "-" & Format(SelMonth, "00")
    
    Dim firstDay As Date
    firstDay = DateSerial(SelYear, SelMonth, 1)
    
    Dim startCol As Integer
    startCol = Weekday(firstDay, vbSunday)
    
    Dim daysInMonth As Integer
    daysInMonth = Day(DateSerial(SelYear, SelMonth + 1, 0))
    
    Dim i As Integer
    Dim dNum As Integer
    
    For i = 1 To 42
        Dim btn As Object
        On Error Resume Next
        Set btn = Me.Controls("btnDay" & i)
        On Error GoTo 0
        If Not btn Is Nothing Then
            dNum = i - startCol + 1
            If dNum >= 1 And dNum <= daysInMonth Then
                btn.caption = dNum
                btn.Visible = True
            Else
                btn.Visible = False
            End If
        End If
    Next i
End Sub

Private Sub SelectDate(dNum As Integer)
    If Not targetCell Is Nothing Then
        targetCell.Value = Format(DateSerial(SelYear, SelMonth, dNum), "YYYY-MM-DD")
        targetCell.NumberFormat = "YYYY-MM-DD"
    End If
    Unload Me
End Sub
